def obj(value: object) -> dict[str, object]:
    assert isinstance(value, dict) and all(isinstance(k, str) for k in value)
    return cast(dict[str, object], value)


def array(value: object) -> list[object]:
    assert isinstance(value, list)
    return cast(list[object], value)


def number(value: object) -> float:
    assert type(value) in (int, float)
    result = float(cast(float, value))
    assert math.isfinite(result)
    return result


def pairs(items: list[tuple[str, object]]) -> dict[str, object]:
    result: dict[str, object] = {}
    for name, value in items:
        assert name not in result, f'duplicate field {name}'
        result[name] = value
    return result


def invalid(token: str) -> object:
    raise ValueError(f'nonfinite JSON token {token}')


def parse(raw: bytes | str) -> object:
    return cast(object, json.loads(raw, object_pairs_hook=pairs, parse_constant=invalid))


def read(path: Path) -> dict[str, object]:
    return obj(parse(path.read_bytes()))


def sha(path: Path) -> str:
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def interpretation(scores: list[float], ids: list[object]) -> tuple[list[float], list[object]]:
    assert scores and len(scores) == len(ids)
    weights = [math.exp(value - max(scores)) for value in scores]
    total = math.fsum(weights)
    return [weight / total for weight in weights], [ids[i] for i in sorted(range(len(scores)), key=scores.__getitem__, reverse=True)]

    assert admission['asset_count'] == len(assets) == 44
    names = [cast(str, item['path']) for item in assets]

        path = ROOT / cast(str, item['path'])
        assert path.stat().st_size == item['bytes'] and sha(path) == item['sha256'], str(path)
    identity = hashlib.sha256()
    model_prefix = '.models/Decision-2.0-Kai-0.6B/'
    model_files = {cast(str, item['path']).removeprefix(model_prefix): cast(str, item['sha256']) for item in assets if cast(str, item['path']).startswith(model_prefix) and not cast(str, item['path']).endswith('/LICENSE')}

    for name, value in sorted(model_files.items()):
        identity.update(name.encode() + b'\0' + bytes.fromhex(value))
    assert 'kai:' + identity.hexdigest() == MODEL_ID
    archive_info = obj(saved['source_archive'])
    archive_path = RESULTS / Path(cast(str, archive_info['path'])).name

    sources = [obj(item) for item in array(read(archive_path)['files'])]
    table = obj(saved['source_sha256'])

    external: list[dict[str, object]] = []
    for item in sources:
        name, text = cast(str, item['name']), cast(str, item['text'])
        assert hashlib.sha256(text.encode()).hexdigest() == item['sha256'] == table[name]
        if name.startswith('external_'):
            relative = {'external_qwen3': 'models/qwen3/modeling_qwen3.py', 'external_sdpa': 'integrations/sdpa_attention.py'}[name]
            matches = list((ROOT / '.venv/lib').glob('*/site-packages/transformers/' + relative))
            assert len(matches) == 1 and sha(matches[0]) == item['sha256']
            external.append({'name': name, 'sha256': item['sha256'], 'path': item['path']})
        else:
            assert sha(ROOT / name) == item['sha256'], name
    assert {item['name'] for item in external} == {'external_qwen3', 'external_sdpa'}
    python_assets = {item['path']: item['sha256'] for item in assets if cast(str, item['path']).endswith('.py')}
    assert python_assets == {item['name']: item['sha256'] for item in sources if not cast(str, item['name']).startswith('external_')}
    source_texts = {cast(str, item['name']): cast(str, item['text']) for item in sources}

        assert probability_error <= 1e-4 and ranking == old_ranking

        max_probability = max(max_probability, probability_error)
        scores_total += len(observed)
        details.append({'id': case['id'], 'selected': ranking[0], 'ranking': ranking, 'correct': is_correct, 'max_abs_logit_difference': logit_error, 'max_abs_probability_difference': probability_error})
    assert cycle['correct'] == correct and cycle['failed'] == 0 and correct >= reference_correct
    drift = {'total': 77, 'comparable':77, 'selected_equal':selected_equal, 'ranking_equal':ranking_equal, 'baseline_correct':reference_correct, 'candidate_correct':correct, 'max_abs_logit_difference':max_logit, 'max_abs_probability_difference':max_probability, 'logits_within_tolerance': True, 'probabilities_within_tolerance':True, 'passed':True}
    assert obj(saved['drift']) == drift

    events = [obj(item) for item in array(saved['dtype_events'])]
    assert [event['role'] for event in events] == ['q_proj', 'gate_proj', 'head'] * 77
    for event in events:

    for owner in ('backbone', 'head'):
        weights = obj(obj(saved['weight_dtypes'])[owner])
        assert set(weights) == {'torch.float32'} and number(weights['torch.float32']) > 0
    memory = obj(saved['cuda_memory'])
    assert all(type(v) is int and number(v) >= 0 for v in memory.values())
    assert number(memory['peak_reserved_bytes']) >= number(memory['peak_allocated_bytes']) >= number(memory['allocated_bytes']) > 0

    assert ready['admission'] == admission
    assert sha(RESULTS/'cuda-asset-preflight-v2-20261005.json') == 'c7c3d933c8d987b0bd95c87bd70b4b6fb1a8e3878b21a490a91c10a8cb99704c'

    return {'passed':True, 'rows':77, 'unique_cases':77, 'scores':scores_total, 'correct':correct, 'reference_correct':reference_correct, 'drift':drift, 'dtype_events':len(events), 'weights':saved['weight_dtypes'], 'memory':memory, 'occupancy_before':saved['occupancy_before'], 'occupancy_after':post, 'parent':parent, 'asset_count':44, 'source_count':len(sources), 'source_sha256':table, 'asset_sha256':{cast(str,item['path']):item['sha256'] for item in assets}, 'terminal_release_probes':probes, 'external_sources':external, 'artifact_sha256':lineage, 'per_case':details, 'limits':['Single cold development cohort; no speed/held-out-quality/production promotion.', 'MATH/IEEE/cleanup policy verified in frozen source; actual GPU kernel dispatch/independent precision-restoration trace not collected.', 'Raw dtype events observe first-layer projections and head; no every-operation trace.', 'Wall records terminal owned-worker cleanup policy; GPU memory is allocator metrics, not total process/system RSS.', 'Occupancy snapshots are not a hardware reservation; no foreign process was signaled.']}


def main() -> None:
    started = time.perf_counter()
    path, expected, review = Path(sys.argv[1]), sys.argv[2], Path(sys.argv[3])
    result = check(path, expected, path.with_suffix('.wall.json'))
    script = Path(__file__).read_bytes()
    script_sha = hashlib.sha256(script).hexdigest()
    archive = review.with_suffix(f'.audit-{script_sha[:16]}.py')
    with archive.open('xb') as stream:
        stream.write(script)
    result.update(captured_utc=datetime.now(timezone.utc).isoformat(), scope='Independent framework-free CPU audit, no model/device allocation', audit_script={'path':str(archive),'sha256':script_sha,'text':script.decode()}, elapsed_seconds=time.perf_counter()-started, frameworks_imported=[name for name in ('torch','transformers','mlx','numpy') if name in sys.modules])
    assert result['frameworks_imported'] == []
    with review.open('x') as stream:
        json.dump(result, stream, indent=2, allow_nan=False)
        stream.write('\n')
    print(json.dumps({'review':str(review),'sha256':sha(review),'rows':result['rows'],'drift':result['drift'],'frameworks_imported':result['frameworks_imported']}))


if __name__ == '__main__':
    main()
