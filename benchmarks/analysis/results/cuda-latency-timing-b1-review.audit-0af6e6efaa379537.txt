def obj(value: object) -> dict[str, object]:
    assert isinstance(value, dict) and all(isinstance(k, str) for k in value)
    return cast(dict[str, object], value)


def array(value: object) -> list[object]:
    assert isinstance(value, list)
    return cast(list[object], value)


def number(value: object) -> float:
    assert type(value) in (int, float)
    result = float(cast(float, value))
    assert math.isfinite(result)
    return result


def unique(items: list[tuple[str, object]]) -> dict[str, object]:
    result: dict[str, object] = {}
    for key, value in items:
        assert key not in result, key
        result[key] = value
    return result


def nonfinite(token: str) -> object:
    raise ValueError(token)


def parse(raw: bytes | str) -> object:
    return cast(
        object, json.loads(raw, object_pairs_hook=unique, parse_constant=nonfinite)
    )


def read(path: Path) -> dict[str, object]:
    return obj(parse(path.read_bytes()))


def sha(path: Path) -> str:
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def interpretation(
    scores: list[float], ids: list[object]
) -> tuple[list[float], list[object]]:
    assert scores and len(scores) == len(ids)
    weights = [math.exp(x - max(scores)) for x in scores]
    total = math.fsum(weights)
    return [x / total for x in weights], [
        ids[i] for i in sorted(range(len(scores)), key=scores.__getitem__, reverse=True)
    ]


def validate_rows(
    rows: list[dict[str, object]], cases: list[dict[str, object]]
) -> None:
    assert [x["id"] for x in rows] == [x["id"] for x in cases]
    for row, case in zip(rows, cases, strict=True):
        assert row["expected"] == case["expected"] and number(row["latency_ms"]) >= 0
        if row["status"] != "ok":
            assert (
                row["prediction"] is None
                and row["correct"] is False
                and isinstance(row["error"], str)
            )
            continue
        assert row["error"] is None
        p = obj(row["prediction"])
        ids = [obj(x)["id"] for x in array(obj(case["request"])["candidates"])]
        scores = [number(x) for x in array(p["scores"])]
        probabilities, ranking = interpretation(scores, ids)
        assert (
            p["model_id"] == MODEL
            and p["ranking"] == ranking
            and p["selected"] == ranking[0]
        )
        assert row["correct"] is (ranking[0] in array(case["expected"]))
        for saved, rebuilt in zip(
            array(p["probabilities"]), probabilities, strict=True
        ):
            x = number(saved)
            assert abs(x - rebuilt) <= 2 * max(math.ulp(x), math.ulp(rebuilt))


def compare(
    rows: list[dict[str, object]], baseline: list[dict[str, object]]
) -> dict[str, object]:
    assert len(rows) == len(baseline) and rows
    comparable = selected = ranked = bc = cc = 0
    ld: list[float] = []
    pd: list[float] = []
    logits_close = probabilities_close = True
    for row, previous in zip(rows, baseline, strict=True):
        assert row["id"] == previous["id"] and row["expected"] == previous["expected"]
        bc += previous["correct"] is True
        cc += row["correct"] is True
        if row["prediction"] is None or previous["prediction"] is None:
            continue
        a, b = obj(previous["prediction"]), obj(row["prediction"])
        comparable += 1
        selected += a["selected"] == b["selected"]
        ranked += a["ranking"] == b["ranking"]
        for x, y in zip(array(a["scores"]), array(b["scores"]), strict=True):
            lhs, rhs = number(x), number(y)
            ld.append(abs(lhs - rhs))
            logits_close &= math.isclose(lhs, rhs, abs_tol=1e-3, rel_tol=1e-3)
        for x, y in zip(
            array(a["probabilities"]), array(b["probabilities"]), strict=True
        ):
            d = abs(number(x) - number(y))
            pd.append(d)
            probabilities_close &= d <= 1e-4
    complete = comparable == len(rows)
    return dict(
        total=len(rows),
        comparable=comparable,
        selected_equal=selected,
        ranking_equal=ranked,
        baseline_correct=bc,
        candidate_correct=cc,
        max_abs_logit_difference=max(ld, default=None),
        max_abs_probability_difference=max(pd, default=None),
        logits_within_tolerance=complete and logits_close,
        probabilities_within_tolerance=complete and probabilities_close,
        passed=complete
        and selected == len(rows)
        and ranked == len(rows)
        and logits_close
        and probabilities_close
        and cc >= bc,
    )


def verify_report(saved: object, rebuilt: dict[str, object]) -> None:
    actual = obj(saved)
    assert set(actual) == set(rebuilt)
    for key, value in rebuilt.items():
        if key == "max_abs_probability_difference" and value is not None:
            # Independently rebuilt CPU softmax can differ by a few libm ULPs.
            assert abs(number(actual[key]) - number(value)) <= 5e-16
        else:
            assert actual[key] == value, (key, actual[key], value)

def close(actual: object, expected: float) -> None:
    assert math.isclose(number(actual), expected, rel_tol=5e-15, abs_tol=5e-16), (
        actual,
        expected,
    )


def quantile(values: list[float], fraction: float) -> float:
    ordered = sorted(values)
    point = (len(values) - 1) * fraction
    lower = math.floor(point)
    upper = math.ceil(point)
    return ordered[lower] + (ordered[upper] - ordered[lower]) * (point - lower)


def rows_of(cycle: object) -> list[dict[str, object]]:
    return [obj(item) for item in array(obj(cycle)["rows"])]


def check_cycle(
    cycle: object, cases: list[dict[str, object]]
) -> list[dict[str, object]]:
    saved = obj(cycle)
    rows = rows_of(cycle)
    if any(row["latency_ms"] is None for row in rows):
        invalid = obj(saved["invalid_timing"])
        for row in rows:
            if row["latency_ms"] is None:
                assert isinstance(invalid.get(f"row:{row['id']}"), str)
        validate_rows(
            [
                dict(row, latency_ms=0) if row["latency_ms"] is None else row
                for row in rows
            ],
            cases,
        )
    else:
        validate_rows(rows, cases)
    assert saved["correct"] == sum(row["correct"] is True for row in rows)
    assert saved["failed"] == sum(row["status"] != "ok" for row in rows)
    assert (
        type(saved["peak_rss_before_bytes"]) is int
        and type(saved["peak_rss_after_bytes"]) is int
    )
    assert number(saved["peak_rss_before_bytes"]) >= 0 and number(
        saved["peak_rss_after_bytes"]
    ) >= number(saved["peak_rss_before_bytes"])
    if saved["latency_ms"] is not None:
        assert number(saved["latency_ms"]) > 0
    else:
        assert isinstance(saved.get("invalid_timing"), dict)
    return rows


def dispatch(trace: object, batch: int, length: int) -> dict[str, object]:
    saved = obj(trace)
    events = [obj(item) for item in array(saved["dispatch"])]
    shape = [batch, 16, length, 128]
    passed = len(events) == 28
    for event in events:
        tensors = [obj(item) for item in array(event["tensors"])]
        flags = obj(event.get("call_flags", {}))
        scale = flags.get("scale")
        passed &= (
            event["operator"] == "aten.scaled_dot_product_attention.default"
            and event.get("selector_boundary") == "public_before_bias_conversion"
            and len(tensors) == 4
        )
        passed &= (
            set(flags) == {"dropout_p", "is_causal", "enable_gqa", "scale"}
            and type(flags.get("dropout_p")) in (int, float)
            and flags.get("dropout_p") == 0
            and flags.get("is_causal") is False
            and flags.get("enable_gqa") is False
        )
        passed &= scale is None or type(scale) is float and number(scale) == 128**-0.5
        if len(tensors) == 4:
            passed &= all(
                item["shape"] == shape
                and item["dtype"] == "torch.float32"
                and item["device"] == "cuda:0"
                and array(item["stride"])[-1] == 1
                for item in tensors[:3]
            )
            mask = tensors[3]
            passed &= (
                mask["shape"] == [1, 1, length, length]
                and mask["dtype"] == "torch.bool"
                and mask["device"] == "cuda:0"
                and mask["stride"] == [length * length, length * length, length, 1]
            )
        assert event.get("converted_bias_strides") == "unknown"
        if scale is not None:
            assert isinstance(event.get("selector_error"), str)
    counts: dict[str, int] = {}
    for item in array(saved["operators"]):
        op = obj(item)
        assert type(op["count"]) is int and number(op["count"]) >= 0
        name = cast(str, op["name"])
        counts[name] = counts.get(name, 0) + op["count"]
    kernels = [obj(item) for item in array(saved["cuda_kernel_events"])]
    for kernel in kernels:
        assert (
            number(kernel["duration_us"]) >= 0 and type(kernel["device_index"]) is int
        )
    attention = [
        cast(str, item["name"])
        for item in kernels
        if "fmha" in cast(str, item["name"]).lower()
    ]
    forbidden = {
        name: count
        for name, count in counts.items()
        if count > 0
        and any(
            term in name
            for term in ("attention_math", "flash_attention", "cudnn_attention")
        )
    }
    passed &= (
        counts.get("aten::_scaled_dot_product_efficient_attention")
        == counts.get("aten::_efficient_attention_forward")
        == len(attention)
        == 28
        and not forbidden
    )
    passed &= saved["cuda_kernel_names_available"] is True and all(
        name.startswith("fmha_cutlassF_f32_aligned_") and "_sm80(" in name
        for name in attention
    )
    return dict(
        passed=passed,
        public_events=len(events),
        efficient_operator_count=counts.get(
            "aten::_scaled_dot_product_efficient_attention", 0
        ),
        efficient_forward_count=counts.get("aten::_efficient_attention_forward", 0),
        fmha_count=len(attention),
        fmha_names=sorted(set(attention)),
        forbidden=forbidden,
        length=length,
        batch=batch,
        precise_internal_arithmetic=(
            "unknown; FastF32 inferred from upstream source only"
        ),
    )

    observer = read(RESULTS / "cuda-attention-source-declaration-v2-20261005.json")

    archive_info = obj(saved["source_archive"])

    source_files = [obj(item) for item in array(read(archive_path)["files"])]

            reports = array(item["drift"])

            else:
                assert not saved["completed"]
            rebuilt_drift[f"{key}:chunk{index}"] = drift
        assert len(ordinary) <= 77 and unknown[key] == ids[len(ordinary) :]
        attempted += len(ordinary)
        failed += sum(row["status"] != "ok" for row in ordinary)
        scored += sum(
            len(array(obj(row["prediction"])["scores"]))

        observations = [obj(value) for value in array(item["ownership"])]

        for index, observation in enumerate(observations):

def main() -> None:
    start = time.perf_counter()
    path, expected, output, declaration, declaration_sha = (
        Path(sys.argv[1]),
        sys.argv[2],
        Path(sys.argv[3]),
        Path(sys.argv[4]),
        sys.argv[5],
    )
    result = audit(path, expected, declaration, declaration_sha)
    source = Path(__file__).read_bytes()
    source_sha = hashlib.sha256(source).hexdigest()
    archive = output.with_suffix(f".audit-{source_sha[:16]}.txt")
    with archive.open("xb") as stream:
        stream.write(source)
    result.update(
        captured_utc=datetime.now(UTC).isoformat(),
        audit_script=dict(path=str(archive), sha256=source_sha, text=source.decode()),
        elapsed_seconds=time.perf_counter() - start,
        frameworks_imported=[
            name
            for name in ("torch", "transformers", "mlx", "numpy")
            if name in sys.modules
        ],
        audit_cap_seconds=60,
    )
    assert result["frameworks_imported"] == []
    with output.open("x") as stream:
        json.dump(result, stream, indent=2, allow_nan=False)
        stream.write("\n")
    print(
        json.dumps(
            dict(
                path=str(output),
                sha256=sha(output),
                attempted_rows=result["attempted_rows"],
                numerical_passed=result["numerical_passed"],
                performance=result["performance"],
            )
        )
    )


if __name__ == "__main__":
    main()
