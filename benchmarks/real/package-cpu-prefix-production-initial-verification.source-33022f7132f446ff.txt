    return hashlib.sha256(data).hexdigest()

def forbidden(name: str) -> bool:
    path = Path(name)
    parts = path.parts
    return (
        any(part.endswith((".trace", ".gputrace")) for part in parts)
        or any(
            part in {".models", ".venv", "venv", "__pycache__", ".env", ".aws"}
            for part in parts
        )
        or name.endswith(".raw.xml")
        or path.suffix.lower()
        in {
            ".safetensors",
            ".pt",
            ".pth",
            ".onnx",
            ".bin",
            ".pyc",
            ".pyo",
        }
    )


COMMAND_LIFETIMES: list[dict[str, object]] = []


def run_owned(
    args: list[str],
    *,
    cwd: Path | str,
    capture_output: bool,
    text: bool,
    timeout: int,
    env: dict[str, str] | None = None,
) -> subprocess.CompletedProcess[str]:
    assert capture_output is True and text is True and 0 < timeout <= 120
    started = time.monotonic()
    child = subprocess.Popen(
        args,
        cwd=cwd,
        env=env,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        start_new_session=True,
    )
    timed_out = False
    try:
        stdout, stderr = child.communicate(timeout=timeout)
    except subprocess.TimeoutExpired:
        timed_out = True
        try:
            os.killpg(child.pid, signal.SIGKILL)
        except ProcessLookupError:
            pass
        stdout, stderr = child.communicate()
    COMMAND_LIFETIMES.append(
        {
            "command": args,
            "pid": child.pid,
            "cap_seconds": timeout,
            "elapsed_seconds": time.monotonic() - started,
            "timed_out": timed_out,
            "exit_code": child.returncode,
            "reaped": child.poll() is not None,
        }
    )
    if timed_out:
        raise RuntimeError(f"Owned subprocess exceeded {timeout}s and was reaped")
    return subprocess.CompletedProcess(args, child.returncode, stdout, stderr)


def command_record(result: subprocess.CompletedProcess[str]) -> dict[str, object]:
    return {
        "command": result.args,
        "exit_code": result.returncode,
        "stdout": result.stdout,
        "stderr": result.stderr,
    }
